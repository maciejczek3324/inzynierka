    def _setup_scene(self):
        self._robot = Articulation(self.cfg.robot_cfg)
        self.scene.articulations["robot"] = self._robot

        self.cfg.terrain.num_envs = self.scene.cfg.num_envs
        self.cfg.terrain.env_spacing = self.scene.cfg.env_spacing
        self._terrain = self.cfg.terrain.class_type(self.cfg.terrain)

        self.scene.clone_environments(copy_from_source=False)

        if self.device == "cpu":
            self.scene.filter_collisions(global_prim_paths=[self.cfg.terrain.prim_path])

        light_cfg = sim_utils.DomeLightCfg(intensity=2000.0, color=(0.75, 0.75, 0.75))
        light_cfg.func("/World/Light", light_cfg)

    def _pre_physics_step(self, actions: torch.Tensor):
        # KAGANIEC 1: Zabezpieczenie przed atakiem kosmicznych akcji z zepsutej polityki
        actions_clamped = torch.clamp(actions.clone(), min=-1.0, max=1.0)
        # actions_clamped = torch.zeros_like(actions)
        self._prev_prev_actions = self._previous_actions.clone()
        self._previous_actions = self._actions.clone()
        self._actions = actions_clamped
        desired_targets = (
                self.cfg.action_scale * self._actions
                + self._robot.data.default_joint_pos
        )

        max_target_delta = 5.24 * self.step_dt

        target_delta = torch.clamp(
            desired_targets - self._current_targets,
            min=-max_target_delta,
            max=max_target_delta,
        )

        self._current_targets = (
                self._current_targets + target_delta
        )
        # self._robot.set_joint_position_target(self._current_targets)

        self._processed_actions = (self.cfg.action_scale * self._actions) + self._robot.data.default_joint_pos

        # Domain Randomization marszu: Zmiana celu co 150 klatek
        if self.num_envs == 1:
            #TRYB Z PADEM (Skrypt play.py)
            pad_commands, _ = self.gamepad.advance()

            # Odczytujemy wartości z gałek
            x_input = pad_commands[0]
            y_input = pad_commands[1]
            yaw_input = pad_commands[5]

            # --- MARTWA STREFA (Lekarstwo na dryfującą gałkę) ---
            # Jeśli gałka jest wychylona na mniej niż 15%, uznajemy, że jest w spoczynku
            if abs(x_input) < 0.15:
                x_input = 0.0
            if abs(y_input) < 0.15:
                y_input = 0.0
            if abs(yaw_input) < 0.15:
                yaw_input = 0.0

            # Przekazujemy przefiltrowany sygnał do robota
            # self._commands[:, 0] = x_input * 0.3
            # self._commands[:, 1] = y_input * 0.3
            # self._commands[:, 2] = yaw_input * 0.3
            self._commands[:, 0] = 0.10
            self._commands[:, 1] = 0.0
            self._commands[:, 2] = 0.0
            # self._commands[:, 0] = 0.3
            # self._commands[:, 1] = 0.0
            # self._commands[:, 2] = 0.0
        else:
            # TRYB TRENINGU
            # env_ids = (self.episode_length_buf % 150 == 0).nonzero(as_tuple=False).flatten()
            # if len(env_ids) > 0:
                # self._commands[env_ids, 0] = (0.15 + torch.rand(len(env_ids), device=self.device) * 0.15)
            self._commands[:, 0] = 0.10
            self._commands[:, 1] = 0.0
            self._commands[:, 2] = 0.0

    def _get_observations(self) -> dict:
        t = self.episode_length_buf * (self.cfg.sim.dt * self.cfg.decimation)
        self._phase = 2.0 * math.pi * self.cfg.gait_frequency * t

        phase_sin = torch.sin(self._phase).unsqueeze(-1)
        phase_cos = torch.cos(self._phase).unsqueeze(-1)

        obs = torch.cat(
            [
                self._robot.data.root_lin_vel_b,
                self._robot.data.root_ang_vel_b,
                self._robot.data.projected_gravity_b,
                self._commands,
                self._robot.data.joint_pos - self._robot.data.default_joint_pos,
                self._robot.data.joint_vel,
                self._actions,
                phase_sin,
                phase_cos,
            ],
            dim=-1,
        )

        # KAGANIEC 2: Zabezpieczenie przed NaN/inf wchodzącym do sieci
        obs = torch.clamp(obs, min=-100.0, max=100.0)
        return {"policy": obs}

    def _get_rewards(self) -> torch.Tensor:
        # --- OBLICZANIE BŁĘDÓW ŚLEDZENIA KOMEND ---
        # lin_vel_x_error = torch.square(self._commands[:, 0] - self._robot.data.root_lin_vel_b[:, 0])
        vx_world = self._robot.data.root_lin_vel_w[:, 0]

        # filtrowana prędkość - nie nagradzamy chwilowego telepania
        self._vx_filtered = (
                0.97 * self._vx_filtered
                + 0.03 * vx_world
        )

        sigma_x = 0.0025

        track_raw = torch.exp(
            -torch.square(
                self._vx_filtered - self._commands[:, 0]
            ) / sigma_x
        )

        stand_raw = torch.exp(
            -torch.square(self._commands[:, 0]) / sigma_x
        )

        track_x_quality = (
            track_raw - stand_raw
            ) / (
                1.0 - stand_raw + 1e-6
            )

        track_x_quality = torch.clamp(
            track_x_quality,
            min=-1.0,
            max=1.0,
        )
        lin_vel_y_error = torch.square(self._commands[:, 1] - self._robot.data.root_lin_vel_b[:, 1])
        yaw_rate_error = torch.square(self._commands[:, 2] - self._robot.data.root_ang_vel_b[:, 2])
        # Jakość śledzenia zadanej prędkości X.
        # Przy vx_cmd = 0.25 i staniu w miejscu wartość będzie prawie zerowa.
        # track_x_quality = torch.exp(-lin_vel_x_error / 0.0025)

        # Osobna marchewka za faktyczne przesuwanie się do przodu.
        # self._command_time += self.step_dt
        #
        # desired_x = (
        #         self._start_root_x
        #         + self._commands[:, 0] * self._command_time
        # )
        #
        # actual_x = self._robot.data.root_pos_w[:, 0]
        #
        # position_error = torch.square(actual_x - desired_x)
        #
        # progress_track = torch.exp(
        #     -position_error / 0.01
        # )
        cmd_x = self._commands[:, 0]

        cmd_x_abs = torch.clamp(
            torch.abs(cmd_x),
            min=0.05,
        )
        # teraz discovery odkrywanie chodu
        forward_signed = torch.clamp(
            self._vx_filtered
            * torch.sign(cmd_x)
            / cmd_x_abs,
            min=0.0, #z -1
            max=1.0,
        )
        base_height_error = torch.square(self._robot.data.root_pos_w[:, 2] - self.cfg.base_height_target)

        # --- HACK PROMOTORA (Sway Allowance) ---
        # 1. Różnica między tym co każemy (np. 0), a tym jak się buja (oś Y)
        vel_y_diff = torch.abs(self._commands[:, 1] - self._robot.data.root_lin_vel_b[:, 1])
        # 2. Tolerancja: 0.15 m/s. Jeśli buja się mniej, błąd wynosi 0. Jak go znosi za bardzo, kara rośnie kwadratowo!
        vel_y_excess = torch.square(torch.clamp(vel_y_diff - 0.15, min=0.0))

        # Tolerancja dla niekontrolowanego obrotu (Yaw) - pozwalamy na minimalne mikro-obroty przy kroku
        yaw_diff = torch.abs(self._commands[:, 2] - self._robot.data.root_ang_vel_b[:, 2])
        yaw_excess = torch.square(torch.clamp(yaw_diff - 0.1, min=0.0))

        # --- Błędy sprzętowe i postawy ---
        action_rate_error = torch.sum(torch.square(self._actions - self._previous_actions), dim=1)
        action_smoothness_error = torch.sum(
            torch.square(self._actions - 2.0 * self._previous_actions + self._prev_prev_actions), dim=1)
        orientation_error = torch.sum(torch.square(self._robot.data.projected_gravity_b[:, :2]), dim=1)
        upright_quality = torch.exp(-orientation_error / 0.10)

        height_quality = torch.exp(-base_height_error / 0.0025)

        locomotion_gate = upright_quality * height_quality
        # --- METRONOM CHODU ---
        if not hasattr(self, "stopa_l_idx"):
            self.stopa_l_idx = self._robot.data.body_names.index("stopaL_1")
            self.stopa_r_idx = self._robot.data.body_names.index("stopaR_1")

        pos_l = self._robot.data.body_pos_w[:, self.stopa_l_idx, :]
        pos_r = self._robot.data.body_pos_w[:, self.stopa_r_idx, :]

        # left_foot_desired_z = torch.clamp(torch.sin(self._phase), min=0.0) * 0.08
        # right_foot_desired_z = torch.clamp(torch.sin(self._phase + math.pi), min=0.0) * 0.08
        #
        # left_foot_z = pos_l[:, 2] - self.scene.env_origins[:, 2]
        # right_foot_z = pos_r[:, 2] - self.scene.env_origins[:, 2]
        #
        # phase_error = torch.abs(left_foot_z - left_foot_desired_z) + torch.abs(right_foot_z - right_foot_desired_z)
        # phase_reward = torch.exp(-phase_error / 0.05)
        left_foot_z = pos_l[:, 2] - self.scene.env_origins[:, 2]
        right_foot_z = pos_r[:, 2] - self.scene.env_origins[:, 2]

        # Metronom określa tylko różnicę wysokości stóp.
        # Nie próbuje wciskać originu stopy dokładnie w z=0.
        # desired_foot_dz = 0.05 * torch.sin(self._phase)
        # actual_foot_dz = left_foot_z - right_foot_z
        #
        # phase_error = torch.square(actual_foot_dz - desired_foot_dz)
        # phase_reward = torch.exp(-phase_error / 0.0025)
        sin_phase = torch.sin(self._phase)

        # Która noga jest aktualnie swing
        left_swing = (sin_phase >= 0.0).float()
        right_swing = 1.0 - left_swing

        # --- 1. UNIESIENIE NOGI SWING ---
        left_clearance = torch.clamp(
            left_foot_z - right_foot_z,
            min=0.0,
        )

        right_clearance = torch.clamp(
            right_foot_z - left_foot_z,
            min=0.0,
        )

        swing_clearance = (
                left_swing * left_clearance
                + right_swing * right_clearance
        )

        clearance_target = 0.035  # 3.5 cm

        clearance_quality = torch.exp(
            -torch.square(swing_clearance - clearance_target) / 0.0004
        )

        # --- 2. NOGA PODPOROWA NIE POWINNA SIĘ ŚLIZGAĆ ---
        vel_l_xy = self._robot.data.body_lin_vel_w[
            :, self.stopa_l_idx, :2
        ]

        vel_r_xy = self._robot.data.body_lin_vel_w[
            :, self.stopa_r_idx, :2
        ]

        stance_slip = (
                left_swing * torch.sum(torch.square(vel_r_xy), dim=1)
                + right_swing * torch.sum(torch.square(vel_l_xy), dim=1)
        )

        stance_quality = torch.exp(
            -stance_slip / 0.02
        )

        # Trochę rewardu za odkrycie kroku nawet na początku,
        # ale pełna nagroda dopiero gdy nadąża za ruchem X.
        forward_gate = torch.clamp(forward_signed, min=0.0, max=1.0)
        gait_gate = 0.25 + 0.75 * forward_gate

        gait_reward = (
                clearance_quality
                * stance_quality
                * gait_gate
        )
        # --- ROZSTAW STÓP (13 cm) ---
        feet_dist = torch.norm(pos_l - pos_r, dim=1)
        stance_error = torch.square(torch.clamp(0.15 - feet_dist, min=0.0))

        # --- SŁOWNIK NAGRÓD I KAR ---
        rewards = {
            "alive_reward": self.cfg.rew_scale_alive * (1.0 - self.reset_terminated.float()),
            "termination_penalty": self.cfg.rew_scale_terminated * self.reset_terminated.float(),

            # MARCHEWKI
            "track_lin_vel_x_exp": self.cfg.rew_scale_track_lin_vel_x_exp * track_x_quality * locomotion_gate,
            "forward_progress_reward":self.cfg.rew_scale_forward_progress * forward_signed,
            "track_lin_vel_y_exp": self.cfg.rew_scale_track_lin_vel_y_exp * torch.exp(-lin_vel_y_error / 0.02),
            "track_yaw_vel_exp": self.cfg.rew_scale_track_yaw_vel_exp * torch.exp(-yaw_rate_error / 0.02),
            "base_height_reward": self.cfg.rew_scale_base_height * torch.exp(-base_height_error / 0.01),
            "phase_sync_reward": self.cfg.rew_scale_phase_sync * gait_reward,

            # KIJE NA ZNOSZENIE (Z aktywną tolerancją Promotora!)
            "lin_vel_y_penalty": self.cfg.rew_scale_lin_vel_y_penalty * vel_y_excess,
            "ang_vel_z_penalty": self.cfg.rew_scale_ang_vel_z_penalty * yaw_excess,

            # KIJE NA CWANIACTWO
            #"phase_sync_penalty": self.cfg.rew_scale_phase_sync * phase_error,
            "feet_stance_penalty": self.cfg.rew_scale_feet_stance * stance_error,
            "orientation_penalty": self.cfg.rew_scale_orientation * orientation_error,

            # OCHRONA SPRZĘTU
            "action_rate_penalty": self.cfg.rew_scale_action_rate * action_rate_error,
            "action_smoothness_penalty": self.cfg.rew_scale_action_smoothness * action_smoothness_error,
            "action_penalty": self.cfg.rew_scale_action_penalty * torch.sum(torch.square(self._actions), dim=1),
            "joint_vel_penalty": self.cfg.rew_scale_joint_vel * torch.sum(torch.abs(self._robot.data.joint_vel), dim=1),
        }

        # Sumowanie
        reward = torch.sum(torch.stack(list(rewards.values())), dim=0)
        reward = torch.clamp(reward, min=-200.0, max=100.0)

        for key, value in rewards.items():
            if key not in self._episode_sums:
                self._episode_sums[key] = torch.zeros(self.num_envs, dtype=torch.float, device=self.device)
            self._episode_sums[key] += value

        return reward

    def _get_dones(self) -> tuple[torch.Tensor, torch.Tensor]:
        base_height = self._robot.data.root_pos_w[:, 2]
        time_out = self.episode_length_buf >= self.max_episode_length - 1

        # Upadek z za niskiej pozycji
        died = base_height < self.cfg.base_height_fall

        # WYKRYWANIE GLITCHY: Jeżeli fizyka wyrzuci go na metr do góry, przerywamy!
        glitched = base_height > 1.0

        tilt_error = torch.sum(
            torch.square(self._robot.data.projected_gravity_b[:, :2]),
            dim=1
        )

        fell_over = tilt_error > 0.75

        died = torch.logical_or(died, fell_over)
        # Ubijamy epizod w obu przypadkach (zapobiega NaN)
        died = torch.logical_or(died, glitched)

        return died, time_out

    def _reset_idx(self, env_ids: torch.Tensor | None):
        if env_ids is None or len(env_ids) == self.num_envs:
            env_ids = self._robot._ALL_INDICES

        self._robot.reset(env_ids)
        super()._reset_idx(env_ids)

        if len(env_ids) == self.num_envs:
            self.episode_length_buf[:] = torch.randint_like(self.episode_length_buf, high=int(self.max_episode_length))

        self._actions[env_ids] = 0.0
        self._previous_actions[env_ids] = 0.0
        self._prev_prev_actions[env_ids] = 0.0
        self._commands[env_ids] = 0.0
        if self.num_envs > 1:
            self._commands[env_ids, 0] = 0.10
            self._commands[env_ids, 1] = 0.0
            self._commands[env_ids, 2] = 0.0

        if hasattr(self, "_current_targets"):
            self._current_targets[env_ids] = self._robot.data.default_joint_pos[env_ids].clone()

        joint_pos = self._robot.data.default_joint_pos[env_ids]
        joint_vel = self._robot.data.default_joint_vel[env_ids]
        default_root_state = self._robot.data.default_root_state[env_ids].clone()
        default_root_state[:, :3] += self._terrain.env_origins[env_ids]
        default_root_state[:, 2] = self.cfg.base_height_target

        self._vx_filtered[env_ids] = 0.0
        self._command_time[env_ids] = 0.0
        self._prev_root_x[env_ids] = default_root_state[:, 0]
        self._start_root_x[env_ids] = default_root_state[:, 0]

        self._robot.write_root_pose_to_sim(default_root_state[:, :7], env_ids)
        self._robot.write_root_velocity_to_sim(default_root_state[:, 7:], env_ids)
        self._robot.write_joint_state_to_sim(joint_pos, joint_vel, None, env_ids)

        extras = dict()
        for key in self._episode_sums.keys():
            episodic_sum_avg = torch.mean(self._episode_sums[key][env_ids])
            extras["Episode_Reward/" + key] = episodic_sum_avg / self.max_episode_length_s
            self._episode_sums[key][env_ids] = 0.0

        self.extras["log"] = dict()
        self.extras["log"].update(extras)
        self.extras["log"]["Episode_Termination/time_out"] = torch.count_nonzero(self.reset_time_outs[env_ids]).item()
