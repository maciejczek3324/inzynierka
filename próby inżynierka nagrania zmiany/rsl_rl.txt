from isaaclab.utils import configclass

from isaaclab_rl.rsl_rl import RslRlOnPolicyRunnerCfg, RslRlPpoActorCriticCfg, RslRlPpoAlgorithmCfg


@configclass
class PPORunnerCfg(RslRlOnPolicyRunnerCfg):
    num_steps_per_env = 32  # ZWIĘKSZONE: Kaczka musi zebrać dane z pełnego kroku przed aktualizacją
    max_iterations = 150000  # Żebyś nie musiał wpisywać w terminalu z palca
    save_interval = 50
    experiment_name = "inzynierka_kaczka_rl_2"  # Zmienione z cartpole!
    empirical_normalization = True  # BARDZO WAŻNE: Automatycznie skaluje obserwacje do równych wartości

    policy = RslRlPpoActorCriticCfg(
        init_noise_std=0.5,
        # MÓZG ROBOTA: Powiększony do standardów lokomocji (było 32x32)
        actor_hidden_dims=[256, 128, 64],
        critic_hidden_dims=[256, 128, 64],
        activation="elu",
    )

    algorithm = RslRlPpoAlgorithmCfg(
        value_loss_coef=1.0,
        use_clipped_value_loss=True,
        clip_param=0.2,
        entropy_coef=0.005,  # ZWIĘKSZONE: Wymusza większą kreatywność w ruchach na początku
        num_learning_epochs=5,
        num_mini_batches=4,
        learning_rate=3.0e-4,
        schedule="adaptive",
        gamma=0.99,
        lam=0.95,
        desired_kl=0.01,
        max_grad_norm=1.0,
    )
